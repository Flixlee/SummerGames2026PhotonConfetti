\begin{frame}{What we learned along the way}
	\begin{columns}[T]
		\column{7.0cm}
		\begin{block}<1->{About the metric}
			\begin{itemize}
				\item the SDES penalizes late signals (finite averaging windows)
			\end{itemize}
		\end{block}
		\begin{block}<1->{About control}
			\begin{itemize}
				\item punctuality matters more than smoothness
				\item filtering needs preview to pay for its delay
			\end{itemize}
		\end{block}
		\column{7.0cm}
		\begin{block}<1->{About the atmosphere}
			\begin{itemize}
				\item turbulent structures are tilted: the top beams see them earlier
				\item beams measuring at different heights must stay synchronous
				\item the apparent transport speed is not simply the mean wind speed
			\end{itemize}
		\end{block}
	\end{columns}
\end{frame}

\begin{frame}{Conclusion and Outlook}

		\end{itemize}
	\end{block}
